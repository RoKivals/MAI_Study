%------------------------ Prefix block
an_prist(Prefix, Combined):-
    get_prefix(L),
    member(Prefix:Combined, L).

%------------------------ Root block
an_kor(Root, Combined):-
    get_root(L),
    member(Root:Combined, L).

%------------------------ Suffix block
an_suff(Suffix, Rod, Plurality):-
    get_suffix(L),
    member(Suffix:Rod:Plurality, L).
