cis(comp(A, B), comp(A, B)):-!.
cis(comp(X, Y), comp(A, B) 'c+' comp(C, D)):-
    X is A + C,
    Y is B + D, !.
cis(comp(X, Y), comp(A, B) 'c-' comp(C, D)):-
    X is A - C,
    Y is B - D, !.
cis(comp(X, Y), comp(A, B) 'c*' comp(C, D)):-
    X is A * C - B * D,
    Y is A * D + B * C, !.
cis(comp(X, Y), comp(A, B) 'c/' comp(C, D)):-
    X is (A*C+B*D)/(C*C + D*D),
    Y is (C*B - A*D)/(C*C + D*D), !.
cis(R, E):-

    cis(A, C),
    cis(B, D),
    T=..[Op, C, D],
